# ex34: Demonstration to obtain the positon at the mouse cursor
```
#
# using external program:  src/c++03/mandelbrot.cc
#
args ndiv=100,ntry=50

@ cmd = "sys ./bin/mandelbrot"

title "key 'ENTR':zoom, 'BS':back, 'f':fine, 'c':coarse, 'ESC':exit"

fmt %.15g
vp 0 0 0 0 1

# Initial range to show Mandelbrot plot
@ n = 0
@ x0[n] = -2
@ x1[n] = 0.5
@ y0[n] = -1.3
@ y1[n] = 1.3

@ zwsize = 0.2 ;# zoom window size

while 1
#  prn "next zwsize=[zwsize] x0=[x0[n]],x1=[x1[n]],y0=[y0[n]],y1=[y1[n]]"
  prn "next zwsize=[zwsize] ndiv=[ndiv] ntry=[ntry]"

# draw Mandelbrot
  [cmd] -ndiv [ndiv] -ntry [ntry] -xr "[x0[n]]:[x1[n]]" -yr "[y0[n]]:[y1[n]]"
  mread x,y,z mandelbrot.dat
  mplot x y z (xm:0 ym:0 mt:cont2)

  getpos x y key; # get x,y position at the mouse cursor and key label

  if [key] == {Mouse_L, Enter}
    if [zwsize] > 1e-6
      ++n
      @ x0[n] = [x-zwsize]
      @ x1[n] = [x+zwsize]
      @ y0[n] = [y-zwsize]
      @ y1[n] = [y+zwsize]
      fbox [x0[n]] [x1[n]] [y0[n]] [y1[n]] (lc:pink lw:2)
      @ zwsize *= 0.2
    else
      prn !! Cannot zoom-in any further (zwsize=[zwsize])
    fi
  elif [key] == Arrow_U
    @ y0[n] += [zwsize]
    @ y1[n] += [zwsize]
  elif [key] == Arrow_D
    @ y0[n] -= [zwsize]
    @ y1[n] -= [zwsize]
  elif [key] == Arrow_L
    @ x0[n] -= [zwsize]
    @ x1[n] -= [zwsize]
  elif [key] == Arrow_R
    @ x0[n] += [zwsize]
    @ x1[n] += [zwsize]
  elif [key] == BackSpace
    if [n] > 0 & [zwsize] < 5
      --n
      @ zwsize *= 5
    else
      prn !! Cannot go back any further (zwsize=[zwsize])
    fi
  elif [key] == Escape
    break
  elif [key] == f  ;# fine
    if [ntry]<2000; @ ntry += 50;  fi
    if [ndiv]<=500; @ ndiv += 100; fi
  elif [key] == c ;# Coarse
    if [ntry]>=100; @ ntry -= 50;  fi
    if [ndiv]>=200; @ ndiv -= 100; fi
  else
    break
  fi
end
```

## How to use this macro

1. Do `exe doc/macro/ex34.tip`

2. Forcus mouse cursor on the graph

3. Enter the command key on the graph

|key | description |
|---|---|
|ENTR key|Zoom in the area around the mouse cursor|
|BS key  |Back to the previous drawing screen|
|ESC key |Exit the macro|
|'f' key |Resolution become fine|
|'c' key |Resolution become coarse|
|UP key  |Shift to up-side|
|DOWN key|Shift to down-side|
|LEFT key|Shift to left-side|
|RIGHT key|Shift to right-side|
