# ex34: Demonstration to obtain the positon at the mouse cursor
```
#
# using external program:  src/c++03/mandelbrot.cc
#
args ndiv=100,ntry=50

@ cmd = "sys ./bin/mandelbrot"

title "key 'ENTR':zoom, 'BS':back, 'f':fine, 'c':coarse, 'ESC':exit"

@ left_click = 1   ;# left button of mouse
@ enter_key = 36   ;# 'ENTR' key
@ back_key  = 22   ;# 'BS' key
@ escape_key = 9   ;# 'ESC' key
@ fine_key   = 41  ;# 'f' key
@ coarse_key = 54  ;# 'c' key
@ up_key = 111     ;# 'UP' key
@ down_key = 116   ;# 'DWN' key
@ left_key = 113   ;# 'LFT' key
@ right_key = 114  ;# 'RGT' key

fmt %.15g
vp 0 0 0 0 1

# Initial range to show Mandelbrot plot
@ n = 0
@ x0[n] = -2
@ x1[n] = 0.5
@ y0[n] = -1.3
@ y1[n] = 1.3

@ zwsize = 0.2 ;# zoom window size

while 1
  prn "next zwsize=[zwsize] x0=[x0[n]],x1=[x1[n]],y0=[y0[n]],y1=[y1[n]]"

# draw Mandelbrot
  [cmd] -ndiv [ndiv] -ntry [ntry] -xr "[x0[n]]:[x1[n]]" -yr "[y0[n]]:[y1[n]]"
  mread x,y,z mandelbrot.dat
  mplot x y z (xm:0 ym:0 mt:cont2)

  getpos x y nkey; # get x,y position at the mouse cursor and key number

  if [nkey]=={[left_click],[enter_key]}
    if [zwsize] > 1e-6
      ++n
      @ x0[n] = [x-zwsize]
      @ x1[n] = [x+zwsize]
      @ y0[n] = [y-zwsize]
      @ y1[n] = [y+zwsize]
      fbox [x0[n]] [x1[n]] [y0[n]] [y1[n]] (lc:pink lw:2)
      @ zwsize *= 0.2
    else
      prn !! Cannot zoom-in any further (zwsize=[zwsize])
    fi
  elif [nkey]==[up_key]
    @ y0[n] += [zwsize]
    @ y1[n] += [zwsize]
  elif [nkey]==[down_key]
    @ y0[n] -= [zwsize]
    @ y1[n] -= [zwsize]
  elif [nkey]==[left_key]
    @ x0[n] -= [zwsize]
    @ x1[n] -= [zwsize]
  elif [nkey]==[right_key]
    @ x0[n] += [zwsize]
    @ x1[n] += [zwsize]
  elif [nkey]==[back_key] 
    if [n] > 0 & [zwsize] < 5
      --n
      @ zwsize *= 5
    else
      prn !! Cannot go back any further (zwsize=[zwsize])
    fi
  elif [nkey]==[escape_key]
    break
  elif [nkey]==[fine_key]
    if [nrty]<2000
      @ ntry += 50
    fi
    if [ndiv]<=500
      @ ndiv = [ntry*2]
    fi
  elif [nkey]==[coarse_key]
    if [nrty]>100
      @ ntry -= 50
    fi
    @ ndiv = [ntry*2]
  else
    break
  fi
end
```

## How to use this macro

1. Do `exe doc/macro/ex34.tip`

2. Forcus mouse cursor on the graph

3. Enter the command key on the graph

|key | description |
|---|---|
|ENTR key|zoom in|
|BS key  |back to previous|
|ESC key |zoom in|
|'f' key |resolution become fine|
|'c' key |resolution become coarse|
|UP key  |shift to up-side|
|DOWN key|shift to down-side|
|LEFT key|shift to left-side|
|RIGHT key|shift to right-side|
